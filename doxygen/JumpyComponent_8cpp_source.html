143 void getIdealSize(int& idealWidth, int& idealHeight) override
144 {
145 auto resultingIdealSize = juce::Rectangle<int>(idealWidth, idealHeight);
146 auto mc = juce::Desktop::getInstance().getComponent(0);
147 if (mc)
148 {
149 auto fBounds = mc->getBounds().toFloat();
150 auto h = fBounds.getHeight();
151 auto w = fBounds.getWidth();
152 if (h > 0.0f && w > 0.0f)
153 {
154 if (h > w)
155 {
156 w = 0.75f * w;
157 h = w;
158 }
159 else
160 {
161 h = 0.75f * h;
162 w = h;
163 }
164
165 resultingIdealSize = juce::Rectangle<float>(w, h).toNearestInt();
166 }
167 }
168
169 if (resultingIdealSize.getWidth() < m_minIdealSize.getWidth() && resultingIdealSize.getHeight() < m_minIdealSize.getHeight())
170 {
171 idealWidth = m_minIdealSize.getWidth();
172 idealHeight = m_minIdealSize.getHeight();
173 }
174 else
175 {
176 idealWidth = resultingIdealSize.getWidth();
177 idealHeight = resultingIdealSize.getHeight();
178 }
179 }

182 {
183 if (m_component)
184 m_component->setBounds(getLocalBounds());
185 }

379void JumpyComponent::timerCallback()
380{
381 auto currentMillisecondsHiRes = juce::Time::getMillisecondCounterHiRes();
382 auto elapsedMillisecondsHiRes = currentMillisecondsHiRes - m_startMillisecondsHiRes;
383
384 auto hours = int(elapsedMillisecondsHiRes / sc_millisInHour);
385 auto overrunMillis = (hours * sc_millisInHour);
386 auto minutes = int(int(elapsedMillisecondsHiRes - overrunMillis) / sc_millisInMin);
387 overrunMillis += (minutes * sc_millisInMin);
388 auto seconds = int(int(elapsedMillisecondsHiRes - overrunMillis) / sc_millisInSec);
389 overrunMillis += (seconds * sc_millisInSec);
390 auto frames = int(int(elapsedMillisecondsHiRes - overrunMillis) / getCurrentFrameIntervalMs());
391
392 auto newTS = TimeStamp(hours, minutes, seconds, frames);
393
394 if (newTS != m_ts)
395 setAndSendTimeCode(newTS);
396}
397
398void JumpyComponent::setStartMilliseconds()
399{
400 m_startMillisecondsHiRes = juce::Time::getMillisecondCounterHiRes(); // get the counter start reference value
401 m_startMillisecondsHiRes -= (m_ts.getHours() * sc_millisInHour) + (m_ts.getMinutes() * sc_millisInMin) + (m_ts.getSeconds() * sc_millisInSec) + (m_ts.getFrames() * getCurrentFrameIntervalMs()); // add any offset from currently set TC value
402}

447void JumpyComponent::oscMessageReceived(const OSCMessage& message)
448{
449 DBG(juce::String(__FUNCTION__) << " " << message.getAddressPattern().toString());
450
451 if (message.getAddressPattern().matches(juce::OSCAddress(juce::String("/") + juce::JUCEApplication::getInstance()->getApplicationName() + "/TS"))
452 && 1 == message.size()
453 && juce::OSCTypes::string == message.begin()->getType())
454 {
455 auto ts = TimeStamp(message.begin()->getString());
456 if (ts.isValid())
457 setAndSendTimeCode(ts);
458 }
459 else
460 {
461 for (auto const& ct : m_customTriggers)
462 {
463 auto& customTriggerDetails = ct.second->getTriggerDetails();
464 auto customTriggerOSCAddress = juce::OSCAddress(customTriggerDetails.m_oscTrigger.getAddressPattern().toString());
465 if (message.getAddressPattern().matches(customTriggerOSCAddress) && customTriggerDetails.m_TS.isValid())
466 setAndSendTimeCode(ct.second->getTriggerDetails().m_TS);
467 }
468 }
469}

471void JumpyComponent::midiMessageReceived()
472{
473 std::vector<juce::MidiMessage> messageQueueCopy;
474 {
475 std::lock_guard<std::mutex> l(m_midiInputMessageQueueMutex);
476 messageQueueCopy = m_midiInputMessageQueue;
477 m_midiInputMessageQueue.clear();
478 }
479
480 for (auto const& midiMesage : messageQueueCopy)
481 {
482 for (auto const& ct : m_customTriggers)
483 {
484 if (ct.second)
485 {
486 auto& customTriggerDetails = ct.second->getTriggerDetails();
487 if (customTriggerDetails.m_midiTrigger.isMatchingCommand(midiMesage) && customTriggerDetails.m_TS.isValid())
488 setAndSendTimeCode(ct.second->getTriggerDetails().m_TS);
489 }
490 }
491 }

495{

519 triggerDetailsXml->setAttribute(JumpyConfiguration::getAttributeName(JumpyConfiguration::AttributeID::IDENT), juce::String(customTrigger.first));
520 triggerDetailsXml->addTextElement(customTrigger.second->getTriggerDetails().toString());
521 customTriggersXml->addChildElement(triggerDetailsXml.release());
522 }
523 m_config->setConfigState(std::move(customTriggersXml), JumpyConfiguration::getTagName(JumpyConfiguration::TagID::CUSTOMTRIGGERS));
524 }
525 }

528void JumpyComponent::onConfigUpdated()
529{
530 auto deviceConfigXml = m_config->getConfigState(JumpyConfiguration::getTagName(JumpyConfiguration::TagID::DEVCONFIG));
531 if (deviceConfigXml)
532 {
533 m_frameRate = deviceConfigXml->getIntAttribute(JumpyConfiguration::getAttributeName(JumpyConfiguration::AttributeID::FRAMERATE));
534
535 m_oscPortNumber = deviceConfigXml->getIntAttribute(JumpyConfiguration::getAttributeName(JumpyConfiguration::AttributeID::OSCPORT));
536
537 auto midiInputDeviceIdentifier = deviceConfigXml->getChildElementAllSubText(JumpyConfiguration::getTagName(JumpyConfiguration::TagID::MIDIINPUT), "");
538 auto midiInputDeviceName = juce::String();
539 for (auto const& midiInputDeviceInfo : m_currentMidiInputDevicesInfos)
540 if (midiInputDeviceInfo.identifier == midiInputDeviceIdentifier)
541 midiInputDeviceName = midiInputDeviceInfo.name;
542 if (midiInputDeviceIdentifier.isNotEmpty() && midiInputDeviceName.isNotEmpty())
543 {
544 for (auto i = getInputDeviceOptionIdRangeStart(); i < getInputDeviceOptionIdRangeEnd(); i++)
545 m_optionsItems[i].second = m_optionsItems[i].first == midiInputDeviceName;
546 openMidiInputDevice(midiInputDeviceIdentifier);
547 }
548 else
549 m_midiInput.reset();
550
551 auto midiOutputDeviceIdentifier = deviceConfigXml->getChildElementAllSubText(JumpyConfiguration::getTagName(JumpyConfiguration::TagID::MIDIOUTPUT), "");
552 auto midiOutputDeviceName = juce::String();
553 for (auto const& midiOutputDeviceInfo : m_currentMidiOutputDevicesInfos)
554 if (midiOutputDeviceInfo.identifier == midiOutputDeviceIdentifier)
555 midiOutputDeviceName = midiOutputDeviceInfo.name;
556 if (midiOutputDeviceIdentifier.isNotEmpty() && midiOutputDeviceName.isNotEmpty())
557 {
558 for (auto i = getOutputDeviceOptionIdRangeStart(); i < getOutputDeviceOptionIdRangeEnd(); i++)
559 m_optionsItems[i].second = m_optionsItems[i].first == midiOutputDeviceName;
560 openMidiOutputDevice(midiOutputDeviceIdentifier);
561 }
562 else
563 m_midiOutput.reset();
564 }
565
566 auto customTriggersXml = m_config->getConfigState(JumpyConfiguration::getTagName(JumpyConfiguration::TagID::CUSTOMTRIGGERS));
567 if (customTriggersXml)
568 {

788void JumpyComponent::sendMessage(TimeStamp ts, int frameRate)
789{
790
791 if (m_timecodeEditor) m_timecodeEditor->setText(ts.toString());
792 DBG(juce::String(__FUNCTION__) << " " << ts.toString());
793
794 if (!m_midiOutput)
795 return;
796
797 unsigned char bytes[] = { 0x7F, 0x7F, 0x01, 0x01, 0x20, 0x00, 0x00, 0x00 };
798 bytes[4] = (ts.getHours() & 0x1F) + ((frameRate << 5) & 0xE0);
799 bytes[5] = static_cast<unsigned char>(ts.getMinutes());
800 bytes[6] = static_cast<unsigned char>(ts.getSeconds());
801 bytes[7] = static_cast<unsigned char>(ts.getFrames());
802 auto midiMessage = juce::MidiMessage::createSysExMessage(bytes, 8);
803
804 m_midiOutput->sendMessageNow(midiMessage);
805}
