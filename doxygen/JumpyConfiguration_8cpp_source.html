106bool JumpyConfiguration::HandleConfigVersionConflict(const JUCEAppBasics::AppConfigurationBase::Version& configVersionFound)
107{
108 if (configVersionFound != JUCEAppBasics::AppConfigurationBase::Version::FromString(JUMPY_CONFIG_VERSION))
109 {
110 auto conflictTitle = "Incompatible configuration version";
111 auto conflictInfo = "The configuration file version detected\ncannot be handled by this version of " + juce::JUCEApplication::getInstance()->getApplicationName();
112#ifdef DEBUG
113 conflictInfo << "\n(Found " + configVersionFound.ToString() + ", expected " + JUMPY_CONFIG_VERSION + ")";
114#endif
115 juce::AlertWindow::showOkCancelBox(juce::MessageBoxIconType::WarningIcon, conflictTitle, conflictInfo, "Reset to default", "Quit", nullptr, juce::ModalCallbackFunction::create([this](int result) {
116 if (1 == result)
117 {
118 ResetToDefault();
119 }
120 else
121 {
122 juce::JUCEApplication::getInstance()->quit();
123 }
124 }));
125
126 return false;
127 }
128 else
129 return true;
130}
