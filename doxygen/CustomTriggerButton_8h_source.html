109 const juce::String toString() const {
110 juce::StringArray timeDigits = {
111 juce::String(getHours()).paddedLeft('0', 2),
112 juce::String(getMinutes()).paddedLeft('0', 2),
113 juce::String(getSeconds()).paddedLeft('0', 2),
114 juce::String(getFrames()).paddedLeft('0', 2)
115 };

129 static TimeStamp fromString(const juce::String& tsString)
130 {
131 auto success = true;
132
133 juce::StringArray time;
134 time.addTokens(tsString, ":", "");
135 if (time.size() != 4)
136 success = false;
137
138 auto ts = TimeStamp(
139 time[0].getIntValue(),
140 time[1].getIntValue(),
141 time[2].getIntValue(),
142 time[3].getIntValue());
143 success = success && ts.isValid();
144
145 if (!success)
146 {
147 juce::AlertWindow::showAsync(
148 juce::MessageBoxOptions()
149 .withMessage("Invalid Timecode value " + tsString + ".")
150 .withButton("Ok")
151 .withIconType(juce::MessageBoxIconType::WarningIcon),
152 nullptr);
153 return {};
154 }
155 else
156 return ts;
157 };
158
159private:
160 int m_hours = 0;
161 int m_minutes = 0;
162 int m_seconds = 0;
163 int m_frames = 0;

261 static TriggerDetails fromString(const juce::String& paramStr)

263 auto sa = juce::StringArray();

266
267 TriggerDetails td;
268 td.m_Name = sa[0];
269 td.m_Colour = juce::Colour::fromString(sa[1]);
270 td.m_TS = TimeStamp::fromString(sa[2]);
271 td.m_oscTrigger.setAddressPattern(sa[3]);
272 if (sa[4].isNotEmpty())
273 td.m_midiTrigger.deserializeFromHexString(sa[4]);
274
275 return td;
276 }
