28CustomTriggerButton::CustomTriggerButton(const juce::String& buttonName)
29 : juce::Button(buttonName)
30{
31 setEnabled(false);
32
33 m_nameLabel = std::make_unique<juce::Label>("Name", buttonName);
34 m_nameLabel->setJustificationType(juce::Justification::centred);
35 m_nameLabel->setInterceptsMouseClicks(false, false);
36 m_tsLabel = std::make_unique<juce::Label>("TS");
37 m_tsLabel->setJustificationType(juce::Justification::centred);
38 m_tsLabel->setInterceptsMouseClicks(false, false);
39 m_settingsButton = std::make_unique<juce::DrawableButton>("SettingsButton", juce::DrawableButton::ButtonStyle::ImageFitted);
40 m_settingsButton->setColour(juce::DrawableButton::ColourIds::backgroundColourId, juce::Colours::transparentBlack);
41 m_settingsButton->setColour(juce::DrawableButton::ColourIds::backgroundOnColourId, juce::Colours::transparentBlack);
42 m_settingsButton->onClick = [=]() { showTriggerSettings(); };
43
44 lookAndFeelChanged();
45}

51void CustomTriggerButton::setTriggerDetails(const CustomTriggerButton::TriggerDetails& triggerDetails)
52{
53 if (!isEnabled())
54 setEnabled(true);
55
56 m_triggerDetails = triggerDetails;
57
58 setColour(juce::TextButton::ColourIds::buttonColourId, m_triggerDetails.m_Colour);
59 setColour(juce::TextButton::ColourIds::buttonOnColourId, m_triggerDetails.m_Colour.brighter());
60
61 m_nameLabel->setText(m_triggerDetails.m_Name, juce::dontSendNotification);
62 m_nameLabel->setFont(m_nameLabel->getFont().withPointHeight(30));
63 addAndMakeVisible(m_nameLabel.get());
64
65 m_tsLabel->setText(m_triggerDetails.m_TS.toString(), juce::dontSendNotification);
66 addAndMakeVisible(m_tsLabel.get());
67
68 addAndMakeVisible(m_settingsButton.get());
69
70 resized();
71
72 if (onDetailsChanged)
73 onDetailsChanged(m_triggerDetails);
74}

97void CustomTriggerButton::mouseDown(const juce::MouseEvent& e)
98{
99 if (!isEnabled())
100 {
101 m_triggerDetails = TriggerDetails(getName(), juce::Colours::cornflowerblue, TimeStamp(),

160 class TriggerSettingsComponent : public juce::Component
161 {
162 public:
169 TriggerSettingsComponent(const TriggerDetails& td, const juce::String& midiInputIdentifier)
170 {
171 m_nameEdit = std::make_unique<JUCEAppBasics::FixedFontTextEditor>();
172 m_nameEdit->setText(td.m_Name);
173 m_nameEdit->setExplicitFocusOrder(0);
174 addAndMakeVisible(m_nameEdit.get());
175
176 m_tcEdit = std::make_unique<JUCEAppBasics::FixedFontTextEditor>();
177 m_tcEdit->setText(td.m_TS.toString());
178 m_tcEdit->setExplicitFocusOrder(0);
179 addAndMakeVisible(m_tcEdit.get());
180
181 m_colourSelector = std::make_unique<juce::ColourSelector>();
182 m_colourSelector->setName("Colour");
183 m_colourSelector->setCurrentColour(td.m_Colour);
184 m_colourSelector->setColour(juce::ColourSelector::backgroundColourId, Colours::transparentBlack);
185 m_colourSelector->setExplicitFocusOrder(0);
186 addAndMakeVisible(m_colourSelector.get());
187
188 m_oscLabel = std::make_unique<juce::Label>("oscLabel", "OSC:");
189 m_oscLabel->setJustificationType(juce::Justification::centredBottom);
190 m_oscLabel->setExplicitFocusOrder(0);
191 addAndMakeVisible(m_oscLabel.get());
192
193 m_oscStringEdit = std::make_unique<JUCEAppBasics::FixedFontTextEditor>();
194 m_oscStringEdit->setText(td.m_oscTrigger.getAddressPattern().toString());
195 m_oscStringEdit->setExplicitFocusOrder(0);
196 addAndMakeVisible(m_oscStringEdit.get());
197
198 m_midiLabel = std::make_unique<juce::Label>("midiLabel", "MIDI:");
199 m_midiLabel->setJustificationType(juce::Justification::centredBottom);
200 m_midiLabel->setExplicitFocusOrder(0);
201 addAndMakeVisible(m_midiLabel.get());
202
203 m_midiLearnButton = std::make_unique<JUCEAppBasics::MidiLearnerComponent>(std::int16_t(-1), JUCEAppBasics::MidiLearnerComponent::AT_Trigger);
204 m_midiLearnButton->setSelectedDeviceIdentifier(midiInputIdentifier);
205 m_midiLearnButton->setCurrentMidiAssi(td.m_midiTrigger);
206 m_midiLearnButton->setExplicitFocusOrder(0);
207 addAndMakeVisible(m_midiLearnButton.get());
208
209 m_okButton = std::make_unique<juce::TextButton>("OkButton");
210 m_okButton->setButtonText("Ok");
211 m_okButton->onClick = [=]() { findParentComponentOfClass<CallOutBox>()->exitModalState(0); };
212 m_okButton->setExplicitFocusOrder(1);
213 addAndMakeVisible(m_okButton.get());
214
215 setSize(300, 530);
216 };
217 ~TriggerSettingsComponent() override
218 {
219 if (onFinished)
220 onFinished(TriggerDetails(m_nameEdit->getText(), m_colourSelector->getCurrentColour(), TimeStamp(m_tcEdit->getText()),
221 juce::OSCMessage(m_oscStringEdit->getText()),
222 m_midiLearnButton->getCurrentMidiAssi()));
223 }
224
225 void resized() override
226 {
227 auto bounds = getLocalBounds();
228
229 m_nameEdit->setBounds(bounds.removeFromTop(30).reduced(5));
230 m_tcEdit->setBounds(bounds.removeFromTop(30).reduced(5));
231 m_okButton->setBounds(bounds.removeFromBottom(30).removeFromRight(bounds.getWidth() / 2).reduced(1));
232 bounds.removeFromBottom(20);
233 auto midiBounds = bounds.removeFromBottom(30);
234 m_midiLabel->setBounds(midiBounds.removeFromLeft(40).reduced(0, 5));
235 m_midiLearnButton->setBounds(midiBounds.reduced(3));
236 auto oscBounds = bounds.removeFromBottom(30);
237 m_oscLabel->setBounds(oscBounds.removeFromLeft(40).reduced(0, 5));
238 m_oscStringEdit->setBounds(oscBounds.reduced(5));
239 m_colourSelector->setBounds(bounds.removeFromBottom(400));
240 }
241
245 std::function<void(const TriggerDetails&)> onFinished;
246
247 private:
248 std::unique_ptr<JUCEAppBasics::FixedFontTextEditor> m_nameEdit;
249 std::unique_ptr<JUCEAppBasics::FixedFontTextEditor> m_tcEdit;
250 std::unique_ptr<juce::ColourSelector> m_colourSelector;
251 std::unique_ptr<juce::Label> m_oscLabel;
252 std::unique_ptr<JUCEAppBasics::FixedFontTextEditor> m_oscStringEdit;
253 std::unique_ptr<juce::Label> m_midiLabel;
254 std::unique_ptr<JUCEAppBasics::MidiLearnerComponent> m_midiLearnButton;
255 std::unique_ptr<juce::TextButton> m_okButton;
256 };
257
258
259 auto tsc = std::make_unique<TriggerSettingsComponent>(m_triggerDetails, m_midiInputDeviceIdentifier);
260 tsc->onFinished = [=](const TriggerDetails& td) { setTriggerDetails(td); repaint(); };
261
262 juce::CallOutBox::launchAsynchronously(std::move(tsc), ((isEnabled() && m_settingsButton) ? m_settingsButton->getScreenBounds() : getScreenBounds()), nullptr);
263}
