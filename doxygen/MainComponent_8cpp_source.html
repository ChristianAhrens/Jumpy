69void MainComponent::resized()
70{
71 auto safety = JUCEAppBasics::iOS_utils::getDeviceSafetyMargins();
72 auto safeBounds = getLocalBounds();
73#if JUCE_IOS
74 safeBounds.reduce(3, 3);
75#endif
76 safeBounds.removeFromTop(safety._top);
77 safeBounds.removeFromBottom(safety._bottom);
78 safeBounds.removeFromLeft(safety._left);
79 safeBounds.removeFromRight(safety._right);
80
81 m_JumpyComponent->setBounds(safeBounds);
82}
